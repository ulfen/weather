    I'd structure it so that GPS is the primary result, and the place name is an optional enhancement.

    async function getCurrentLocation() {
        const coords = await getGpsCoordinates();

        // These are the authoritative location values.
        const { latitude, longitude } = coords;

        // Start with coordinates so we always have something useful to display.
        const location = {
            latitude,
            longitude,
            name: formatCoordinates(latitude, longitude)
        };

        // Reverse geocoding is optional.
        try {
            const response = await fetch(
                `https://nominatim.openstreetmap.org/reverse` +
                `?lat=${latitude}` +
                `&lon=${longitude}` +
                `&format=json`
            );

            if (!response.ok) {
                throw new Error(`Reverse geocoding failed: ${response.status}`);
            }

            const data = await response.json();

            location.name = getPlaceName(data.address);
        } catch (error) {
            console.warn("Could not determine place name:", error);
            // Keep the coordinate name we already have.
        }

        return location;
    }


    function getGpsCoordinates() {
        return new Promise((resolve, reject) => {
            if (!navigator.geolocation) {
                reject(new Error("Geolocation is not supported by this browser."));
                return;
            }

            navigator.geolocation.getCurrentPosition(
                position => {
                    resolve({
                        latitude: position.coords.latitude,
                        longitude: position.coords.longitude
                    });
                },
                error => {
                    reject(error);
                },
                {
                    enableHighAccuracy: true,
                    timeout: 10000,
                    maximumAge: 300000
                }
            );
        });
    }


    function getPlaceName(address) {
        return (
            address.city ||
            address.town ||
            address.village ||
            address.municipality ||
            address.county ||
            null
        );

    function formatCoordinates(latitude, longitude) {

        return (
